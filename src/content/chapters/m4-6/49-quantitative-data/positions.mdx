คะแนนแปดสิบอาจสูงมากในวิชาหนึ่งแต่ธรรมดาในอีกวิชา ต้องรู้ตำแหน่งเมื่อเทียบกับคนอื่น ไม่ใช้คะแนนดิบเพียงอย่างเดียว

บอกตำแหน่งได้สองวิธี วิธีแรกนับลำดับ เช่นคะแนนนี้สูงกว่าคนประมาณร้อยละ $70$ ของห้อง ซึ่งเป็นควอร์ไทล์และเปอร์เซ็นไทล์ที่เริ่มไว้ในบทที่ 26 วิธีที่สองวัดระยะ เช่นคะแนนนี้สูงกว่าค่าเฉลี่ยสองเท่าของส่วนเบี่ยงเบนมาตรฐาน ซึ่งคือคะแนนมาตรฐาน หน้านี้ทำทั้งสองวิธี แล้วใช้คะแนนมาตรฐานเทียบคะแนนต่างวิชาที่ยากง่ายไม่เท่ากัน

## ตำแหน่งหลังเรียงข้อมูล

หน้านี้ใช้ตำแหน่ง $Q_j=j(n+1)/4$ และ $P_k=k(n+1)/100$ ในข้อมูลเรียงจากน้อยไปมาก ถ้าตำแหน่งไม่เป็นจำนวนเต็มให้แทรกระหว่างค่าข้างเคียงตามสัดส่วน วิธีนี้ใช้เมื่อดัชนีอยู่ในช่วงข้อมูล วิธีซอฟต์แวร์และหนังสือบางแหล่งต่างกัน จึงต้องระบุข้อตกลงและไม่ผสมสูตร

ควอร์ไทล์แบ่งข้อมูลเป็นสี่ส่วน **เดไซล์** แบ่งเป็นสิบส่วน มีตำแหน่ง $D_k=k(n+1)/10$ และ**เปอร์เซ็นไทล์**แบ่งเป็นร้อยส่วน สามแบบนี้คือเรื่องเดียวกันที่แบ่งละเอียดต่างกัน $Q_1=P_{25}$ $Q_2=D_5=P_{50}$ ซึ่งก็คือมัธยฐาน และ $Q_3=P_{75}$ เปอร์เซ็นไทล์ที่ $k$ อ่านว่าข้อมูลประมาณร้อยละ $k$ มีค่าไม่เกินค่านี้

<Example n={1}>

ข้อมูลเรียง $2,4,6,8,10,12,14$ หา $Q_1,Q_3$

<Steps reveal>
1. $n=7$ ตำแหน่ง $Q_1=(7+1)/4=2$ และ $Q_3=6$
2. ได้ $Q_1=4,Q_3=12$ พิสัยระหว่างควอร์ไทล์ $8$
</Steps>
</Example>

{/* check: [2,4,6,8,10,12,14][1]==4 and [2,4,6,8,10,12,14][5]==12 */}

## ตำแหน่งเศษส่วนและการแทรกค่า

สูตรตำแหน่งอาจให้ค่าระหว่างดัชนี ถ้าใช้ข้อตกลงแทรกเชิงเส้น ให้เศษของตำแหน่งเป็นสัดส่วนระยะระหว่างสองค่าที่ติดกัน วิธีนี้เป็นนิยามเชิงคำนวณ ไม่ได้แปลว่ามีคนอยู่ที่ดัชนีครึ่งคน และผลอาจต่างจากวิธีควอร์ไทล์อื่นที่ใช้แบ่งครึ่งข้อมูล

<Example n={2}>

ข้อมูลเรียง $2,4,8,10,14$ ใช้ตำแหน่ง $Q_1=(n+1)/4$ หา $Q_1$

<Steps reveal>
1. ตำแหน่งคือ $6/4=1.5$ อยู่ครึ่งทางระหว่างตัวที่หนึ่งกับสอง
2. แทรกค่า $2+0.5(4-2)=3$
3. จึงได้ $Q_1=3$ ภายใต้ข้อตกลงนี้ แม้ไม่มีข้อมูลเท่ากับสามจริง
</Steps>
</Example>

{/* check: 2+Rational(1,2)*(4-2)==3 */}

<Example n={3}>

คะแนนสอบเก็บเต็ม $50$ ของนักเรียน $15$ คนเรียงแล้วเป็น

$$
18,21,23,25,26,28,30,31,33,34,36,38,41,44,47
$$

จงหา $P_{70}$ และ $D_3$ แล้วแปลความหมาย

<Steps reveal>
1. ข้อมูลเรียงแล้ว $n=15$ ตำแหน่งของ $P_{70}$ คือ $\tfrac{70(15+1)}{100}=11.2$ อยู่ระหว่างตัวที่ $11$ กับ $12$ ค่อนไปทางตัวที่ $11$
2. นับจากซ้าย ตัวที่ $11$ คือ $36$ ตัวที่ $12$ คือ $38$ ห่างกัน $2$ คะแนน เศษ $0.2$ ของตำแหน่งบอกให้เดินไป $0.2$ ของระยะนี้ ได้ $P_{70}=36+0.2(2)=36.4$
3. ตำแหน่งของ $D_3$ คือ $\tfrac{3(15+1)}{10}=4.8$ ตัวที่ $4$ คือ $25$ ตัวที่ $5$ คือ $26$ ได้ $D_3=25+0.8(1)=25.8$
4. แปลผล นักเรียนประมาณร้อยละ $70$ ได้ไม่เกิน $36.4$ คะแนน และประมาณร้อยละ $30$ ได้ไม่เกิน $25.8$ คะแนน คนที่ได้ $38$ จึงอยู่ในกลุ่มราวร้อยละ $30$ ที่คะแนนสูงที่สุดของกลุ่มนี้
</Steps>
</Example>

{/* check: (lambda d: d[10] + Rational(1,5)*(d[11]-d[10]) == Rational(182,5) and d[3] + Rational(4,5)*(d[4]-d[3]) == Rational(129,5))([18,21,23,25,26,28,30,31,33,34,36,38,41,44,47]) and Rational(70*16,100)==Rational(56,5) and Rational(3*16,10)==Rational(24,5) */}

<Note title="ตรวจวิธีคิด">
ถ้าต้องทำงานเทียบกับซอฟต์แวร์ ให้บอกชื่อวิธีหาตำแหน่ง ไม่ปรับคำตอบให้ตรงเครื่องมือโดยไม่ตรวจนิยาม โดยเฉพาะชุดเล็กที่วิธีต่างกันอาจให้ค่าค่อนข้างต่าง
</Note>

## แผนภาพกล่องและค่าผิดปกติ

บทที่ 26 วาดแผนภาพกล่องจากค่าห้าค่า คือค่าต่ำสุด $Q_1$ $Q_2$ $Q_3$ และค่าสูงสุด แล้วลากเส้นจากกล่องไปถึงค่าต่ำสุดและสูงสุดเสมอ ปัญหาคือถ้ามีค่าหนึ่งหลุดไปไกลมาก เส้นจะยาวจนดูเหมือนข้อมูลกระจายสม่ำเสมอไปถึงตรงนั้น ทั้งที่จริงมีแค่ค่าเดียว ม.ปลาย จึงแยกค่าที่อยู่ไกลผิดสังเกตออกมาด้วยเกณฑ์ที่ใช้พิสัยระหว่างควอร์ไทล์เป็นไม้บรรทัด

<Def>
ให้ $\text{IQR}=Q_3-Q_1$ ข้อมูลที่น้อยกว่า $Q_1-1.5\,\text{IQR}$ หรือมากกว่า $Q_3+1.5\,\text{IQR}$ เรียกว่า**ค่าผิดปกติ** ในแผนภาพกล่องให้วาดค่าผิดปกติเป็นจุดแยก แล้วลากเส้นจากกล่องไปถึงข้อมูลตัวที่ไกลที่สุดที่ยังไม่ผิดปกติ
</Def>

ใช้ $\text{IQR}$ แทนส่วนเบี่ยงเบนมาตรฐานเพราะ $\text{IQR}$ คำนวณจากครึ่งกลางของข้อมูล ค่าผิดปกติที่เรากำลังหาจึงไม่ไปขยายไม้บรรทัดที่ใช้ตัดสินตัวมันเอง

<Example n={4}>

คะแนนสอบย่อยของนักเรียน $11$ คนเรียงแล้วเป็น $4,11,12,13,14,15,15,16,17,18,19$ มีค่าผิดปกติหรือไม่ แล้วอธิบายว่าแผนภาพกล่องหน้าตาเป็นอย่างไร

<Steps reveal>
1. $n=11$ ตำแหน่ง $Q_1=\tfrac{12}{4}=3$ $Q_2=6$ $Q_3=9$ เป็นจำนวนเต็มทั้งหมด อ่านได้ $Q_1=12$ $Q_2=15$ $Q_3=17$
2. $\text{IQR}=17-12=5$ คูณ $1.5$ ได้ระยะเผื่อ $7.5$ คะแนน
3. เกณฑ์ล่าง $12-7.5=4.5$ เกณฑ์บน $17+7.5=24.5$
4. คะแนน $4$ น้อยกว่า $4.5$ จึงเป็นค่าผิดปกติ ส่วนคะแนนสูงสุด $19$ ไม่ถึง $24.5$ จึงไม่มีค่าผิดปกติด้านบน
5. วาดกล่องจาก $12$ ถึง $17$ เส้นมัธยฐานที่ $15$ เส้นขวาลากถึง $19$ เส้นซ้ายลากถึง $11$ ซึ่งเป็นข้อมูลตัวที่น้อยที่สุดที่ไม่ผิดปกติ แล้ววาดจุดแยกที่ $4$
</Steps>
</Example>

## คะแนนมาตรฐาน

<Thm>
สำหรับประชากรที่ $\sigma>0$ คะแนนมาตรฐานคือ $z=(x-\mu)/\sigma$ บอกว่าค่านั้นห่างจากค่าเฉลี่ยกี่ส่วนเบี่ยงเบนมาตรฐาน ค่า $z$ บวกอยู่เหนือค่าเฉลี่ยและลบอยู่ต่ำกว่า
</Thm>

สูตรนี้ทำสองขั้นที่เห็นในหน้า 49.3 ลบค่าเฉลี่ยเพื่อย้ายจุดศูนย์ไปไว้ที่ค่าเฉลี่ย แล้วหารด้วยส่วนเบี่ยงเบนมาตรฐานเพื่อเปลี่ยนหน่วยเป็นจำนวนเท่าของส่วนเบี่ยงเบนมาตรฐาน ผลคือคะแนนมาตรฐานของทั้งชุดมีค่าเฉลี่ย $0$ และส่วนเบี่ยงเบนมาตรฐาน $1$ เสมอ ไม่ว่าข้อมูลเดิมจะมีหน่วยอะไร ข้อมูลต่างวิชาหรือต่างหน่วยจึงมาอยู่บนไม้บรรทัดเดียวกันได้

คะแนนมาตรฐานคำนวณได้โดยไม่จำเป็นต้องแจกแจงปกติ แต่การแปลงเป็นความน่าจะเป็นด้วยตารางปกติต้องมีสมมติฐานเพิ่มตามบทที่ 50

<Example n={5}>

คะแนน $75$ ในวิชาที่เฉลี่ย $60$ ส่วนเบี่ยงเบนมาตรฐาน $10$ ได้คะแนนมาตรฐานเท่าไร

<Steps reveal>
1. $z=(75-60)/10=1.5$
2. จึงอยู่เหนือค่าเฉลี่ยหนึ่งจุดห้าส่วนเบี่ยงเบนมาตรฐาน ยังไม่ได้บอกเปอร์เซ็นไทล์โดยไม่รู้การแจกแจง
</Steps>
</Example>

{/* check: Rational(75-60,10)==Rational(3,2) */}

<Think>

ได้ $z=1$ แปลว่าอยู่เปอร์เซ็นไทล์ที่หนึ่งหรือไม่

<Reveal>
ไม่ใช่ หนึ่งคือระยะหนึ่งส่วนเบี่ยงเบนมาตรฐานเหนือค่าเฉลี่ย เปอร์เซ็นไทล์ต้องอาศัยรูปร่างการแจกแจงหรือข้อมูลลำดับสะสม
</Reveal>
</Think>

## เทียบคะแนนสองวิชาด้วยคะแนนมาตรฐาน

<Example n={6}>

ปลายภาคนี้ฝนได้คณิตศาสตร์ $72$ คะแนน ซึ่งห้องเฉลี่ย $60$ ส่วนเบี่ยงเบนมาตรฐาน $8$ และได้ภาษาอังกฤษ $80$ คะแนน ซึ่งห้องเฉลี่ย $74$ ส่วนเบี่ยงเบนมาตรฐาน $6$ ฝนทำวิชาใดได้ดีกว่าเมื่อเทียบกับเพื่อนในห้อง

<Steps reveal>
1. คะแนนดิบภาษาอังกฤษสูงกว่า แต่ข้อสอบสองวิชายากง่ายไม่เท่ากัน จึงต้องเทียบแต่ละวิชากับเพื่อนในวิชานั้นก่อน
2. คณิตศาสตร์ $z=\tfrac{72-60}{8}=1.5$ คือสูงกว่าค่าเฉลี่ยหนึ่งจุดห้าเท่าของส่วนเบี่ยงเบนมาตรฐาน
3. ภาษาอังกฤษ $z=\tfrac{80-74}{6}=1$ สูงกว่าค่าเฉลี่ยหนึ่งเท่าของส่วนเบี่ยงเบนมาตรฐาน
4. ฝนทำคณิตศาสตร์ได้ดีกว่าเมื่อเทียบกับห้อง แม้คะแนนดิบจะน้อยกว่า เพราะห้องทำคณิตศาสตร์ได้น้อยกว่าโดยรวม การได้ $12$ คะแนนเหนือค่าเฉลี่ยในวิชานี้จึงโดดเด่นกว่า $6$ คะแนนเหนือค่าเฉลี่ยในวิชาที่ทุกคนได้ใกล้ๆ กัน
</Steps>
</Example>

{/* check: Rational(72-60,8)==Rational(3,2) and Rational(80-74,6)==1 */}

เมื่อต้องรวมคะแนนหลายวิชาเพื่อจัดอันดับ เช่นคัดตัวแทนไปแข่งขัน การบวกคะแนนดิบทำให้วิชาที่ส่วนเบี่ยงเบนมาตรฐานใหญ่มีน้ำหนักมากเกินไป เพราะในวิชานั้นคะแนนห่างกันได้มาก ผลรวมคะแนนมาตรฐานให้ทุกวิชามีน้ำหนักเท่ากัน

<Example n={7}>

วิชาคณิตศาสตร์มีค่าเฉลี่ย $55$ ส่วนเบี่ยงเบนมาตรฐาน $12$ วิชาภาษาอังกฤษมีค่าเฉลี่ย $70$ ส่วนเบี่ยงเบนมาตรฐาน $5$ ต้นได้คณิตศาสตร์ $79$ ภาษาอังกฤษ $72$ ส่วนเมย์ได้คณิตศาสตร์ $61$ ภาษาอังกฤษ $80$ ถ้าคัดคนที่มีผลรวมคะแนนมาตรฐานสูงกว่า ควรเลือกใคร

<Steps reveal>
1. ผลรวมคะแนนดิบ ต้นได้ $79+72=151$ เมย์ได้ $61+80=141$ ถ้าดูแค่นี้ต้นนำ $10$ คะแนน
2. คะแนนมาตรฐานของต้น คณิตศาสตร์ $\tfrac{79-55}{12}=2$ ภาษาอังกฤษ $\tfrac{72-70}{5}=0.4$ รวม $2.4$
3. คะแนนมาตรฐานของเมย์ คณิตศาสตร์ $\tfrac{61-55}{12}=0.5$ ภาษาอังกฤษ $\tfrac{80-70}{5}=2$ รวม $2.5$
4. เมย์มีผลรวมคะแนนมาตรฐานสูงกว่า จึงควรเลือกเมย์ ต้นได้เปรียบในคะแนนดิบเพราะคณิตศาสตร์มีส่วนเบี่ยงเบนมาตรฐานใหญ่ คะแนนที่ห่างจากค่าเฉลี่ยมากจึงพาผลรวมขึ้นไปได้มาก ส่วนภาษาอังกฤษที่ทุกคนได้ใกล้กัน การนำเพื่อน $10$ คะแนนถือว่าเก่งมากแล้ว
</Steps>
</Example>

คะแนนมาตรฐานยังเป็นประตูไปสู่บทที่ 50 เมื่อข้อมูลมีรูปร่างแบบโค้งระฆัง ค่า $z$ จะบอกได้ว่ามีคนประมาณร้อยละเท่าไรอยู่ต่ำกว่าคะแนนนั้น ซึ่งเชื่อมวิธีวัดตำแหน่งสองแบบในหน้านี้เข้าด้วยกัน

## สรุป

- เรียงข้อมูลก่อนหาตำแหน่ง
- บอกวิธีคำนวณควอร์ไทล์และการแทรกค่า เดไซล์และเปอร์เซ็นไทล์ใช้หลักเดียวกัน
- ค่าที่น้อยกว่า $Q_1-1.5\,\text{IQR}$ หรือมากกว่า $Q_3+1.5\,\text{IQR}$ เป็นค่าผิดปกติ วาดเป็นจุดแยกและต้องตรวจสอบ ไม่ใช่ลบทิ้ง
- คะแนนมาตรฐานวัดระยะสัมพัทธ์ ไม่ใช่ร้อยละโดยตรง
- เทียบหรือรวมคะแนนต่างวิชาด้วยคะแนนมาตรฐาน ไม่ใช่คะแนนดิบ
